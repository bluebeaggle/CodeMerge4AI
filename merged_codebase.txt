// ## frontend/js/chart_client.js

/**
 * 독립적인 TradingView 차트 인스턴스를 관리하는 래퍼 클래스
 * Grid 레이아웃 변경 시 자동으로 크기를 맞추는 ResizeObserver를 포함합니다.
 * @class ChartWidget
 */
class ChartWidget {
    /**
     * @param {HTMLElement} container - 차트가 렌더링될 DOM 요소
     * @param {string} symbol - 심볼명 (ex: BTC_USDT)
     * @param {string} timeframe - 타임프레임 (ex: 1m)
     */
    constructor(container, symbol, timeframe) {
        this.container = container;
        this.symbol = symbol;
        this.timeframe = timeframe;
        
        // 차트 초기화
        this.chart = LightweightCharts.createChart(this.container, {
            width: this.container.clientWidth,
            height: this.container.clientHeight,
            layout: {
                backgroundColor: '#131722', // var(--bg-color-base)
                textColor: '#d1d4dc',
            },
            grid: {
                vertLines: { color: '#2a2e39' },
                horzLines: { color: '#2a2e39' },
            },
            crosshair: { mode: LightweightCharts.CrosshairMode.Normal },
            timeScale: { timeVisible: true, secondsVisible: false },
        });

        this.candleSeries = this.chart.addCandlestickSeries({
            upColor: '#26a69a',
            downColor: '#ef5350',
            borderVisible: false,
            wickUpColor: '#26a69a',
            wickDownColor: '#ef5350'
        });

        this._setupResizeObserver();
    }

    /**
     * 컨테이너 크기 변경 감지 (Grid 레이아웃 분할 시 필수)
     * @private
     */
    _setupResizeObserver() {
        this.resizeObserver = new ResizeObserver(entries => {
            if (entries.length === 0 || entries[0].target !== this.container) return;
            const newRect = entries[0].contentRect;
            // 차트 리사이즈 수행 (UI 블로킹 최소화)
            requestAnimationFrame(() => {
                this.chart.applyOptions({ 
                    width: newRect.width, 
                    height: newRect.height 
                });
            });
        });
        this.resizeObserver.observe(this.container);
    }

    /**
     * @param {Array} historyData - 과거 캔들 배열
     */
    setData(historyData) {
        this.candleSeries.setData(historyData);
    }

    /**
     * @param {Object} candle - 실시간 틱 데이터
     */
    update(candle) {
        // TODO: 향후 백엔드에서 심볼/타임프레임을 함께 내려주면 여기서 필터링 로직 추가
        this.candleSeries.update(candle);
    }

    destroy() {
        this.resizeObserver.disconnect();
        this.chart.remove();
        this.container.innerHTML = '';
    }
}

/**
 * 서버와의 HTTP API 및 WebSocket 통신을 전담하는 데이터 서비스 클래스
 * @class DataClient
 */
class DataClient {
    constructor() {
        this.ws = null;
        this.subscribers = []; // WS 데이터를 수신할 콜백 리스트 (Observer Pattern)
    }

    /**
     * 특정 심볼과 타임프레임의 과거 캔들 데이터를 가져옵니다.
     * (백엔드 API가 이 URL 구조를 지원하도록 곧 수정할 예정입니다)
     * * @param {string} symbol - ex) "BTC_USDT"
     * @param {string} timeframe - ex) "1m", "5m"
     * @returns {Promise<Array>} 캔들 데이터 배열
     */
    async fetchHistory(symbol, timeframe) {
        try {
            // SSoT: 엔드포인트 규칙 통일
            const response = await fetch(`/api/history/${symbol}/${timeframe}`);
            if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
            
            const data = await response.json();
            return data;
        } catch (error) {
            console.error(`[DataClient] History fetch failed for ${symbol}(${timeframe}):`, error);
            return []; // UI 브레이크 방지 (Defensive)
        }
    }

    /**
     * 실시간 차트 업데이트를 위한 WebSocket을 연결합니다.
     */
    connectWS() {
        const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
        const wsUrl = `${protocol}//${window.location.host}/ws/chart`;

        this.ws = new WebSocket(wsUrl);

        this.ws.onopen = () => console.info("[DataClient] WebSocket Connected");
        
        this.ws.onmessage = (event) => {
            try {
                const data = JSON.parse(event.data);
                // 등록된 모든 구독자(차트들)에게 데이터 브로드캐스트
                this.subscribers.forEach(callback => callback(data));
            } catch (error) {
                console.error("[DataClient] WS Message Parse Error:", error);
            }
        };

        this.ws.onclose = () => {
            console.warn("[DataClient] WebSocket Disconnected. Reconnecting in 3s...");
            setTimeout(() => this.connectWS(), 3000); // 자동 재연결
        };
    }

    /**
     * 실시간 데이터를 받을 콜백을 등록합니다.
     * @param {Function} callback 
     */
    subscribe(callback) {
        this.subscribers.push(callback);
    }
}

/**
 * 전체 애플리케이션 상태 및 차트 생명주기를 관리하는 메인 컨트롤러
 */
class AppController {
    constructor() {
        this.charts = []; // 생성된 ChartWidget 인스턴스 배열
        this.gridContainer = document.getElementById('chart-grid-container');
        this.layoutIndex = 0;
        this.layouts = ['single', 'split-h', 'split-v', 'quad'];
        
        this._bindUIEvents();
        this._initApp();
    }

    async _initApp() {
        // 1. 초기 단일 차트 생성 (기본값: BTC_USDT, 1m)
        await this.addChart('BTC_USDT', '1m');

        // 2. 웹소켓 연결 및 글로벌 구독 설정
        window.DataClient.connectWS();
        window.DataClient.subscribe((tickData) => {
            // 수신된 틱 데이터를 활성화된 모든 차트 인스턴스에 전달
            this.charts.forEach(chart => chart.update(tickData));
        });
    }

    _bindUIEvents() {
        const btnSplit = document.getElementById('btn-split-layout');
        const btnAdd = document.getElementById('btn-add-chart');

        if (btnSplit) {
            btnSplit.addEventListener('click', () => this.toggleLayout());
        }

        if (btnAdd) {
            btnAdd.addEventListener('click', () => {
                // 임시로 하드코딩된 심볼 추가. 향후 UI 모달창 등에서 선택받도록 확장.
                this.addChart('BTC_USDT', '5m'); 
            });
        }
    }

    /**
     * 새로운 차트를 Grid 컨테이너에 추가합니다.
     * @param {string} symbol 
     * @param {string} timeframe 
     */
    async addChart(symbol, timeframe) {
        if (this.charts.length >= 4) {
            alert("최대 4개의 차트만 띄울 수 있습니다.");
            return;
        }

        // 1. DOM 컨테이너 생성
        const chartDiv = document.createElement('div');
        chartDiv.className = 'chart-pane';
        this.gridContainer.appendChild(chartDiv);

        // 2. ChartWidget 인스턴스화
        const chartWidget = new ChartWidget(chartDiv, symbol, timeframe);
        this.charts.push(chartWidget);

        // 3. 레이아웃 자동 조정 (차트 갯수에 맞춰 Grid 변경)
        this._autoAdjustLayout();

        // 4. 초기 히스토리 데이터 페칭 및 렌더링
        const historyData = await window.DataClient.fetchHistory(symbol, timeframe);
        if (historyData.length > 0) {
            chartWidget.setData(historyData);
        }
    }

    /**
     * 레이아웃 순환 토글 (단일 -> 가로분할 -> 세로분할 -> 4분할)
     */
    toggleLayout() {
        this.layoutIndex = (this.layoutIndex + 1) % this.layouts.length;
        const targetLayout = this.layouts[this.layoutIndex];
        
        // 이전에 작성한 ui_controller.js 의 레이아웃 변경 기능 호출
        if (window.AppUI) {
            window.AppUI.setChartLayout(targetLayout);
        }
    }

    /**
     * 차트 갯수에 따라 레이아웃을 지능적으로 변경 (Defensive)
     * @private
     */
    _autoAdjustLayout() {
        const count = this.charts.length;
        if (count === 1) this.layoutIndex = 0; // single
        else if (count === 2) this.layoutIndex = 1; // split-h
        else if (count >= 3) this.layoutIndex = 3; // quad

        if (window.AppUI) {
            window.AppUI.setChartLayout(this.layouts[this.layoutIndex]);
        }
    }
}

// ## frontend/js/ui_controller.js

/**
 * UI 레이아웃 및 탭 전환을 관리하는 Controller 클래스
 * 비즈니스 로직(데이터 페칭 등)은 포함하지 않으며 오직 View 상태만 관리합니다.
 */
class UIController {
    /**
     * @param {string} navSelector - 네비게이션 버튼 컨테이너 셀렉터
     * @param {string} viewSelector - 뷰 섹션 셀렉터
     */
    constructor(navSelector, viewSelector) {
        this.navButtons = document.querySelectorAll(`${navSelector} .tab-btn`);
        this.views = document.querySelectorAll(viewSelector);
        
        this._initEvents();
    }

    /**
     * 초기 이벤트 바인딩 (메인 스레드 블로킹 방지)
     * @private
     */
    _initEvents() {
        if (!this.navButtons.length || !this.views.length) {
            console.error("[UIController] UI 요소를 찾을 수 없습니다. (DOM 로드 확인 필요)");
            return;
        }

        this.navButtons.forEach(btn => {
            btn.addEventListener('click', (e) => this.switchTab(e.currentTarget));
        });
    }

    /**
     * 탭 전환 로직
     * @param {HTMLElement} selectedBtn - 클릭된 버튼 요소
     */
    switchTab(selectedBtn) {
        const targetId = selectedBtn.getAttribute('data-target');
        if (!targetId) return;

        // 1. 버튼 상태 업데이트
        this.navButtons.forEach(btn => btn.classList.remove('active'));
        selectedBtn.classList.add('active');

        // 2. 뷰 상태 업데이트
        this.views.forEach(view => {
            if (view.id === targetId) {
                view.classList.add('active');
            } else {
                view.classList.remove('active');
            }
        });

        // 3. 차트 리사이즈 이벤트 트리거 (화면 렌더링 최적화)
        // 숨겨져 있던 차트가 다시 나타날 때 크기를 재계산해야 합니다.
        window.dispatchEvent(new Event('resize'));
    }

    /**
     * 차트 그리드 레이아웃 변경
     * @param {string} layoutType - 'single', 'split-h', 'split-v', 'quad'
     */
    setChartLayout(layoutType) {
        const gridContainer = document.getElementById('chart-grid-container');
        if (!gridContainer) return;

        // 기존 레이아웃 클래스 제거 및 새 클래스 부여
        gridContainer.className = `grid-layout-${layoutType}`;
        window.dispatchEvent(new Event('resize')); // 차트 크기 재계산
    }
}

// 전역 싱글톤 인스턴스 (DOMContentLoaded 이후 초기화)
document.addEventListener('DOMContentLoaded', () => {
    window.AppUI = new UIController('#main-nav', '.view-section');
